### Problem. 印出 1 到 n

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n = 0;

    cout << "請輸入 n：";
    cin >> n;

    for (int i = 1; i <= n; i++)
    {
        cout << i << endl;
    }

    return 0;
}

```

### Problem. 計算 1 加到 n

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n, sum = 0;
    cin >> n;

    for (int i = 1; i <= n; i++)
    {
        sum += i;
    }

    cout << sum << endl;
    return 0;
}
```

### Problem. 偶數加總

```cpp
#include <iostream>
using namespace std;

int main()
{
    int n = 0;
    int sum = 0;

    cout << "請輸入 n：";
    cin >> n;

    for (int i = 1; i <= n; i++)
    {
        if (i % 2 == 0)
        {
            sum = sum + i;
        }
    }

    cout << "偶數總和：" << sum << endl;
    return 0;
}

```
