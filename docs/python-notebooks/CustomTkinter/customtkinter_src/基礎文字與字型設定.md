```py
import customtkinter as ctk


### 建立主視窗
app = ctk.CTk() # 建立 CustomTkinter 主視窗
app.title("Label 範例 A") # 設定視窗標題
app.geometry("350x200") # 設定視窗大小，寬 350、高 200，單位為像素


### 建立大標題
title_label = ctk.CTkLabel(
    app, # 指定 Label 放在主視窗 app 裡面
    text="歡迎使用本系統", # 設定 Label 顯示的文字
    font=("微軟正黑體", 24, "bold"), # 設定字型、字體大小與粗體
    text_color="#DBDDDF" # 設定文字顏色，可使用十六進位色碼
)

title_label.pack(pady=20) # 將 Label 放入視窗，上下增加 20 像素的間距


### 建立說明文字
sub_label = ctk.CTkLabel(
    app, # 指定 Label 放在主視窗 app 裡面
    text="請先登入帳號後繼續", # 設定 Label 顯示的文字
    font=("微軟正黑體", 14), # 設定字型為微軟正黑體、字體大小為 14
    text_color="gray" # 設定文字顏色為灰色
)

sub_label.pack() # 將 Label 放入視窗


### 啟動視窗
app.mainloop() # 啟動事件迴圈，讓視窗持續顯示並等待使用者操作
```
